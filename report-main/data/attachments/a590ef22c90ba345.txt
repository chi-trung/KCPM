pm.test('status code is success', function () {
  pm.expect([200, 201, 400, 401, 403, 422]).to.include(pm.response.code);
});
if (pm.response.code === 200 || pm.response.code === 201) {
  const body = pm.response.json();
  const complaintId = body.data?.id || body.id || body.data?.complaintId;
  if (complaintId) {
    pm.environment.set('complaintId', complaintId);
  }
}
