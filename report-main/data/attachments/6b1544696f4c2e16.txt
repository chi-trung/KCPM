// Auto-set managementToken = adminToken (they are the same role)
if (!pm.environment.get("managementToken") || pm.environment.get("managementToken") === "") {
    var adminTk = pm.environment.get("adminToken") || pm.variables.get("adminToken") || "";
    if (adminTk) {
        pm.environment.set("managementToken", adminTk);
    }
}

var requestName = pm.info.requestName || '';
if (requestName === 'GET Health' && pm.collectionVariables.get('verificationInitialized') !== 'true') {
  pm.collectionVariables.set('verificationPassCount', '0');
  pm.collectionVariables.set('verificationFailCount', '0');
  pm.collectionVariables.set('verificationInitialized', 'true');
}
