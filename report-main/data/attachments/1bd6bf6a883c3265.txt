var requestName = pm.info.requestName || '';
if (requestName === 'GET Jira Sync Summary') {
  return;
}

function expectedCodes(name) {
  if (/^GET /.test(name)) {
    if (/By ID|Detail|Profile/.test(name)) {
      return [200, 404];
    }
    if (/Roles|Health|Public|All Categories|Stats|Summary|Overview|My Reports|My Complaints|Notifications|Unread Count|Leaderboard|Waste Types|Reward Rules|Available Collectors|Task Progress|User Stats|Analytics/.test(name)) {
      return [200];
    }
    return [200, 404];
  }
  if (/^POST /.test(name)) {
    return [200, 201, 202, 204, 400, 401, 403, 404, 409, 422];
  }
  if (/^PUT /.test(name) || /^PATCH /.test(name)) {
    return [200, 204, 400, 401, 403, 404, 409, 422];
  }
  if (/^DELETE /.test(name)) {
    return [200, 204, 400, 401, 403, 404];
  }
  return [200, 201, 204, 400, 401, 403, 404, 409, 422];
}

var acceptable = expectedCodes(requestName);
var passed = acceptable.includes(pm.response.code);
var passCount = parseInt(pm.collectionVariables.get('verificationPassCount') || '0', 10);
var failCount = parseInt(pm.collectionVariables.get('verificationFailCount') || '0', 10);

if (passed) {
  pm.collectionVariables.set('verificationPassCount', String(passCount + 1));
} else {
  pm.collectionVariables.set('verificationFailCount', String(failCount + 1));
}
